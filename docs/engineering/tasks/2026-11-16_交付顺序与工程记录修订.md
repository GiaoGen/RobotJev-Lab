### 完整文档复核命令

在仓库根目录执行以下 PowerShell 命令，检查九份主文档及本任务记录。

```powershell
@'
from pathlib import Path
import json
import re
import unicodedata

root = Path.cwd()
main = [root / "Cross_Embodiment_Decision_Lab_PRD_v1.1.md",
        *sorted((root / "docs/implementation").glob("*.md"))]
record = root / "docs/engineering/tasks/2026-11-16_交付顺序与工程记录修订.md"
files = [*main, record]
texts = {path: path.read_text(encoding="utf-8") for path in files}
errors = []
links = 0
examples = 0
fence = chr(96) * 3

def anchors(text):
    result = set()
    for line in text.splitlines():
        if line.startswith("#"):
            title = re.sub(r"^#+\s+", "", line).strip().lower()
            title = "".join(c for c in title if c in "-_ " or
                            unicodedata.category(c)[0] in "LN")
            result.add(title.replace(" ", "-"))
    return result

for path, text in texts.items():
    if text.count(fence) % 2:
        errors.append(f"{path.name}: code fence")
    for sample in re.findall(fence + r"json\s*\n(.*?)\n" + fence, text, re.S):
        json.loads(sample)
        examples += 1
    for target in re.findall(r"\]\(([^)]+)\)", text):
        if target.startswith(("https://", "http://")):
            continue
        links += 1
        filename, separator, anchor = target.partition("#")
        destination = (path.parent / filename).resolve()
        if not destination.exists():
            errors.append(f"{path.name}: {target}")
        elif separator and anchor not in anchors(destination.read_text(encoding="utf-8")):
            errors.append(f"{path.name}: anchor {target}")
    width = None
    fenced = False
    for line in text.splitlines():
        if line.startswith(fence):
            fenced = not fenced
        if fenced:
            continue
        if line.startswith("|"):
            count = len(line.split("|"))
            if width is not None and count != width:
                errors.append(f"{path.name}: table width")
            width = count
        else:
            width = None

master = next(text for path, text in texts.items() if path.name.startswith("00_"))
expected = {f"FR-{n:03}" for n in range(1, 41)}
assert set(re.findall(r"\| (FR-\d{3}) \|", master)) == expected
assert set(re.findall(r"\*\*(FR-\d{3})｜", texts[main[0]])) == expected
assert "V1.2" in texts[main[0]]
for path in main[1:]:
    assert "V0.2" in texts[path]
    assert "任务结束后的" in texts[path] and "六部分" in texts[path]
combined = "\n".join(texts[path] for path in main)
defined = set(re.findall(r"\| ([A-G]\d{2}) \|", combined))
referenced = set(re.findall(r"\b[A-G]\d{2}\b", combined))
assert referenced <= defined
hours = 0
for line in master.splitlines():
    if line.startswith("| 第 ") and "小时" in line:
        hours += sum(map(int, re.findall(r"\d+", line.split("|")[3])))
assert hours == 200
assert "M1 交付时 A01—A04" not in combined
assert "3D 工作台从第 1 周开始形成" not in combined
assert "步骤 4 建立浏览器工作台" not in combined
print(json.dumps({"files": len(files), "requirements": len(expected),
                  "test_ids": len(defined), "local_links": links,
                  "json_examples": examples, "planned_hours": hours,
                  "errors": errors}, ensure_ascii=False))
raise SystemExit(1 if errors else 0)
'@ | python -X utf8 -
```
